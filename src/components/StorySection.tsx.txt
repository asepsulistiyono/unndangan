import "./StorySection.css";

import { useWedding } from "../lib/WeddingContext";

type StoryItem = {
  year?: string;
  title?: string;
  titleEn?: string;
  text?: string;
  textEn?: string;
};

export default function StorySection() {
  const {
    data,
    language,
  } = useWedding();

  const storyItems =
    Array.isArray(data.story)
      ? (data.story as StoryItem[])
      : [];

  return (
    <section
      id="story"
      className="story-section"
    >
      <div className="story-heading">
        <p className="story-eyebrow">
          {language === "en"
            ? "Our Story"
            : "Our Story"}
        </p>

        <h2>
          {language === "en"
            ? "Our Story"
            : "Kisah Kami"}
        </h2>

        <p className="story-description">
          {language === "en"
            ? "Four chapters that brought us to the altar — from an unexpected meeting to the vows we are about to make."
            : "Empat bab yang membawa kami ke altar — dari pertemuan yang tak disengaja hingga janji yang akan segera diikrarkan."}
        </p>
      </div>

      <div className="story-list">
        {storyItems.map(
          (item, index) => {
            const title =
              language === "en"
                ? item.titleEn ||
                  item.title ||
                  ""
                : item.title || "";

            const text =
              language === "en"
                ? item.textEn ||
                  item.text ||
                  ""
                : item.text || "";

            return (
              <article
                className="story-card"
                key={`story-${item.year || index}-${index}`}
              >
                <div className="story-year">
                  {item.year || ""}
                </div>

                <div className="story-card-content">
                  <h3>{title}</h3>
                  <p>{text}</p>
                </div>
              </article>
            );
          }
        )}
      </div>
    </section>
  );
}
