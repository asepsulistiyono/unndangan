type GiftAccount = {
  bank?: string;
  number?: string;
  holder?: string;
};

type GiftSectionProps = {
  data: {
    gifts?: GiftAccount[];
    giftAddress?: string;
  };
};

export default function GiftSection({
  data,
}: GiftSectionProps) {
  const gifts = data.gifts ?? [];
  const giftAddress = data.giftAddress?.trim() ?? "";

  if (gifts.length === 0 && !giftAddress) {
    return null;
  }

  return (
    <section
      id="kado"
      className="bg-pine-950 px-5 py-16 text-ivory sm:px-8"
    >
      <div className="mx-auto max-w-3xl">
        <p className="text-center text-[10px] font-bold uppercase tracking-[0.35em] text-gold-400">
          Tanda Kasih
        </p>

        <h2 className="mt-3 text-center font-display text-3xl font-light italic text-ivory">
          Kado Pernikahan
        </h2>

        <p className="mx-auto mt-4 max-w-xl text-center text-sm leading-7 text-sage-300/75">
          Doa restu Anda merupakan hadiah yang paling berarti bagi kami.
          Namun, jika ingin memberikan tanda kasih, Anda dapat menggunakan
          informasi berikut.
        </p>

        {gifts.length > 0 && (
          <div className="mt-8 grid gap-4 sm:grid-cols-2">
            {gifts.map((gift, index) => (
              <div
                key={`${gift.bank ?? "kado"}-${index}`}
                className="border border-gold-500/25 bg-pine-800/50 p-5 text-center"
              >
                <p className="text-[10px] font-bold uppercase tracking-[0.25em] text-gold-400">
                  {gift.bank || "Transfer"}
                </p>

                {gift.number && (
                  <p className="mt-3 break-all font-display text-xl text-ivory">
                    {gift.number}
                  </p>
                )}

                {gift.holder && (
                  <p className="mt-2 text-xs text-sage-300/75">
                    Atas nama {gift.holder}
                  </p>
                )}
              </div>
            ))}
          </div>
        )}

        {giftAddress && (
          <div className="mt-5 border border-gold-500/25 bg-pine-800/50 p-5 text-center">
            <p className="text-[10px] font-bold uppercase tracking-[0.25em] text-gold-400">
              Alamat Pengiriman Kado
            </p>

            <p className="mt-3 whitespace-pre-line text-sm leading-7 text-sage-200">
              {giftAddress}
            </p>
          </div>
        )}
      </div>
    </section>
  );
}
